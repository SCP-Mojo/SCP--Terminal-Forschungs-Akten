    async function importPDF(event) {
    const file = event.target.files[0];
    const statusTag = document.getElementById('pdfStatus');
    if (!file || file.type !== 'application/pdf') {
    alert('Bitte wähle eine gültige PDF-Datei aus.');
    return;
    }
    statusTag.style.color = '#ffaa00';
    statusTag.innerText = '> LESE PDF-DATEN EIN UND ANALYSIERE FELDER...';
    try {
    const arrayBuffer = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
    let fullText = '';
    for (let i = 1; i <= pdf.numPages; i++) {
    const page = await pdf.getPage(i);
    const content = await page.getTextContent();
    const pageText = content.items.map(item => item.str).join(' ');
    fullText += pageText + '\n';
    }
    // Falls noch keine ID vergeben wurde, aus Datei ermitteln
    const idField = document.getElementById('newId');
    if (!idField.value) {
    const fileNameClean = file.name.replace('.pdf', '').toUpperCase();
    idField.value = `PDF-${fileNameClean}`;
    }
    // TEXT IN SCHLÜSSELWÖRTER PARSEN
    parsePDFTextToFields(fullText);
    statusTag.style.color = '#00ff66';
    statusTag.innerText = `> IMPORT ERFOLGREICH (${pdf.numPages} SEITE(N) AUF FELDER VERTEILT)`;
    } catch (error) {
    console.error('PDF-Import Fehler:', error);
    statusTag.style.color = '#ff3333';
    statusTag.innerText = '> FEHLER BEIM EINLESEN DER PDF-DATEI.';
    }
    }
    /* --- PARSER FUNKTION (ORDNET DEN TEXT DEN FELDERN ZU) --- */
    function parsePDFTextToFields(text) {
    // Zuordnung der Regex-Muster zu den Feld-IDs
    const fieldMappings = [
    { id: 'newUntersuchung', keys: ['UNTERSUCHUNGSPROTOKOLL', 'UNTERSUCHUNG', 'PROTOKOLL', 'BESCHREIBUNG'] },
    { id: 'newAussereAnatomie', keys: ['ÄUSSERE ANATOMIE', 'AUSSRERE ANATOMIE', 'AUSSENANATOMIE'] },
    { id: 'newInnereAnatomie', keys: ['INNERE ANATOMIE', 'INNENANATOMIE'] },
    { id: 'newBlutanalyse', keys: ['BLUTANALYSE', 'BLUTUNTERSUCHUNG', 'BLUT'] },
    { id: 'newGenetik', keys: ['GENETISCHE ANALYSE', 'GENETIK', 'DNA', 'DNS'] },
    { id: 'newTodesursache', keys: ['TODESURSACHE', 'AUTOPSIE', 'NEUTRALISIERUNG'] },
    { id: 'newGefahr', keys: ['GEFAHRENEINSCHÄTZUNG', 'GEFAHRENSTUFE', 'BEDROHUNG'] },
    { id: 'newSchwachstellen', keys: ['SCHWACHSTELLEN', 'ANGRIFFSPUNKTE', 'MÖGLICHE SCHWACHSTELLEN UND ANGRIFFSPUNKTE'] },
    { id: 'newSchlussbemerkung', keys: ['SCHLUSSBEMERKUNG', 'FAZIT', 'BEMERKUNG'] }
    ];
    // Alle Schlüsselwörter für das Aufteilen sammeln
    let allKeys = [];
    fieldMappings.forEach(m => allKeys.push(...m.keys));
    let foundAny = false;
    fieldMappings.forEach(mapping => {
    let extractedContent = '';
    for (let key of mapping.keys) {
    // Erstellt eine Regex, die nach dem Wort sucht und alles bis zum nächsten bekannten Schlüsselwort nimmt
    const otherKeysPattern = allKeys.filter(k => k !== key).join('|');
    const regex = new RegExp(`(?:${key})[:\\s]*([\\s\\S]*?)(?=(?:${otherKeysPattern})[:\\s]|$|---)`, 'i');
    const match = text.match(regex);
    if (match && match[1].trim().length > 0) {
    extractedContent = match[1].trim();
    foundAny = true;
    break;
    }
    }
    if (extractedContent) {
    document.getElementById(mapping.id).value = extractedContent;
    }
    });
    // Falls keine Abschnitte/Keywords erkannt wurden, schreib alles ins Untersuchungsprotokoll
    if (!foundAny) {
    document.getElementById('newUntersuchung').value = text.trim();
    }
    }
